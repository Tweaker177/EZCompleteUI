// ── Helpers ───────────────────────────────────────────────────────────────────

function corsHeaders() {
  return {
    "Access-Control-Allow-Origin":  "*",
    "Access-Control-Allow-Headers": "authorization, content-type",
    "Content-Type":                 "application/json",
  };
}

// Keep enough to recognize the request — first sentence, max 120 chars.
function truncatePrompt(prompt: unknown): string | null {
  if (!prompt || typeof prompt !== "string") return null;
  const cap = 120;
  const search = prompt.slice(0, cap);
  const match  = search.match(/[.?!\n]/);
  if (match && match.index !== undefined && match.index > 30) {
    return prompt.slice(0, match.index + 1);
  }
  return prompt.length <= cap ? prompt : prompt.slice(0, cap) + "…";
}

function jsonResponse(body: object, status = 200) {
  return new Response(JSON.stringify(body), { status, headers: corsHeaders() });
}

async function fetchSubscription(
  supabase: ReturnType<typeof createClient>,
  userId: string
) {
  const { data, error } = await supabase
    .from("subscriptions")
    .select("coins_balance, tier, status, current_period_end, has_ever_purchased")
    .eq("user_id", userId)
    .order("updated_at", { ascending: false })
    .limit(1)
    .maybeSingle();
  return { data, error };
}

// ── Cost estimation helpers ───────────────────────────────────────────────────

// Compute exact image API cost using model + quality + size.
// Also adds prompt text token cost and (for edits) image input token cost.
// Falls back to approximate flat cost if model/quality/size not provided.
function estimateImageApiCost(
  feature: string,
  qty: number,
  imgModel?: string,
  imgQuality?: string,
  imgSize?: string,
  promptTokens?: number,
  isEdit?: boolean
): number {
  const model   = imgModel ?? "gpt-image-2.5-flare";
  // Normalize quality — "auto" maps to this model's highest supported tier
  // ("max" for the 2.5 pair, "high" for everything else), matching
  // ez-image's own conservative assumption (OpenAI resolves "auto" to
  // whichever tier it judges the prompt needs; charging the safe upper
  // bound here avoids this estimate under-predicting what ez-image will
  // actually charge, which would let a request through as "affordable"
  // that then fails for real).
  const highestTier = MODELS_SUPPORTING_XHIGH_MAX.has(model) ? "max" : "high";
  const quality = (imgQuality === "auto" || !imgQuality) ? highestTier : imgQuality;
  const size    = imgSize ?? "1024x1024";
  const ptok    = promptTokens ?? 80; // estimate ~80 tokens if not provided

  // Exact per-image flat fee
  let perImageCost = IMAGE_COST_EXACT[model]?.[quality]?.[size] ?? null;

  if (perImageCost == null) {
    // Fallback: use old approximate rates by feature tier. dall-e-3
    // removed — dropped from this app entirely.
    const approx: Record<string, number> = {
      image_low: 0.006, image_medium: 0.013, image_high: 0.053,
    };
    perImageCost = approx[feature] ?? 0.013;
  }

  // Text input token cost for the prompt (charged per image)
  const textRate   = IMAGE_TEXT_INPUT_PER_1K[model] ?? 0.005;
  const promptCost = (ptok / 1000) * textRate * qty;

  // Image input token cost for edit mode (the attached reference image)
  const imageInputRate = IMAGE_INPUT_TOKEN_PER_1K[model] ?? 0.010;
  const editCost = isEdit ? (EDIT_IMAGE_INPUT_TOKENS / 1000) * imageInputRate * qty : 0;

  return perImageCost * qty + promptCost + editCost;
}

// Compute chat API cost using actual input/output token split and model name.
function estimateChatApiCost(
  feature: string,
  model: string | undefined,
  inputTokens: number,
  outputTokens: number
): number {
  const inRate  = (model ? CHAT_INPUT_PER_1K[model]  : null) ?? CHAT_TIER_INPUT_PER_1K[feature]  ?? 0.000150;
  const outRate = (model ? CHAT_OUTPUT_PER_1K[model] : null) ?? CHAT_TIER_OUTPUT_PER_1K[feature] ?? 0.000600;
  return (inputTokens / 1000) * inRate + (outputTokens / 1000) * outRate;
}

// Compute Sora API cost from feature key and model.
function estimateSoraApiCost(feature: string, soraModel?: string): number {
  const secs     = SORA_DURATION[feature] ?? 10;
  const model    = soraModel ?? (feature.includes("_pro") ? "sora-2-pro" : "sora-2");
  const costPerS = SORA_COST_PER_SEC[model] ?? 0.10;
  return secs * costPerS;
}

function computeCostPer100Coins(apiCostUsd: number, coinsCharged: number): number | null {
  if (coinsCharged <= 0) return null;
  return Math.round((apiCostUsd / coinsCharged) * 100 * 10000) / 10000;
}

// ── Main handler ──────────────────────────────────────────────────────────────

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 204, headers: corsHeaders() });
  }

  try {
    const authHeader = req.headers.get("Authorization");
    if (!authHeader) return jsonResponse({ allowed: false, reason: "No auth" }, 401);

    const supabase = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
    );

    const jwt = authHeader.replace("Bearer ", "");
    const { data: { user }, error: userError } = await supabase.auth.getUser(jwt);
    if (userError || !user) {
      return jsonResponse({ allowed: false, reason: "Invalid token" }, 401);
    }

    // Capture IP and email for audit log — stored on every usage row.
    // IP comes from Cloudflare/Supabase proxy headers; falls back to direct connection.
    const ipAddress = req.headers.get("cf-connecting-ip")
                   ?? req.headers.get("x-forwarded-for")?.split(",")[0].trim()
                   ?? req.headers.get("x-real-ip")
                   ?? null;
    const userEmail = user.email ?? null;

    const body   = await req.json();
    const action: string = body.action ?? "check";

    // ── action: refund_tokens ─────────────────────────────────────────────────
    if (action === "refund_tokens") {
      // NEUTERED (see this file's changelog): this used to credit
      // estimatedCoins - actualCoins back to the user's balance, on the
      // premise that "check" had deducted estimatedCoins upfront. Now that
      // "check" is read-only (see that action's own comment), there's
      // nothing real to refund here — crediting anything would be handing
      // out free coins for a charge that never happened. ez-chat already
      // performs its own complete, correct deduction/overage reconciliation
      // internally (see its own changelog) and writes its own authoritative
      // ez_usage_log row with real token counts and cost, so this action's
      // entire original purpose is now redundant with what ez-chat already
      // does correctly on its own. Kept as a safe no-op — returns success
      // with the real unchanged balance — rather than removed outright, so
      // any existing client call site doesn't need to change immediately;
      // safe to remove the client-side call entirely as cleanup.
      const { data: sub } = await supabase
        .from("subscriptions")
        .select("coins_balance")
        .eq("user_id", user.id)
        .order("updated_at", { ascending: false })
        .limit(1)
        .maybeSingle();
      return jsonResponse({
        success:  true,
        refunded: 0,
        balance:  sub?.coins_balance ?? 0,
      });
    }

    // ── action: refund_flat — direct flat-coin refund, symmetric to "deduct" ──
    // For callers (br-ai) that deduct an exact, pre-known coin amount via
    // "deduct" rather than an estimated token count. refund_tokens above is
    // shaped specifically for the chat token-estimate reconciliation flow
    // (ez-chat) — it requires tier/estimated_tokens/actual_tokens and
    // computes the refund as their difference, which doesn't fit a flat
    // deduction at all. This is just the direct inverse of "deduct": credit
    // back exactly the given amount.
    if (action === "refund_flat") {
      const { coins, feature, reason } = body;
      if (coins == null || coins <= 0) {
        return jsonResponse({ success: false, reason: "Missing or invalid coins" }, 400);
      }

      const { data: sub } = await supabase
        .from("subscriptions")
        .select("coins_balance")
        .eq("user_id", user.id)
        .order("updated_at", { ascending: false })
        .limit(1)
        .maybeSingle();

      const newBalance = (sub?.coins_balance ?? 0) + coins;
      await supabase
        .from("subscriptions")
        .update({ coins_balance: newBalance, updated_at: new Date().toISOString() })
        .eq("user_id", user.id);

      await supabase.from("coin_transactions").insert({
        user_id:       user.id,
        amount:        coins,
        direction:     "credit",
        feature:       feature ?? "refund",
        description:   reason ?? `Refund: ${coins} coins`,
        balance_after: newBalance,
      });

      return jsonResponse({ success: true, refunded: coins, balance: newBalance });
    }

    // ── action: complete_usage — called after image/sora API returns ───────────
    // Updates an ez_usage_log row with actual results (images returned, etc.)
    if (action === "complete_usage") {
      const { log_id, images_returned, error_text } = body;
      if (!log_id) return jsonResponse({ success: false, reason: "Missing log_id" }, 400);

      const update: Record<string, unknown> = {
        status: error_text ? "error" : "complete",
      };
      if (images_returned != null) update.images_returned = images_returned;
      if (error_text)              update.error_text      = error_text;

      await supabase
        .from("ez_usage_log")
        .update(update)
        .eq("id", log_id)
        .eq("user_id", user.id);

      return jsonResponse({ success: true });
    }

    // ── Fetch subscription (all other actions need it) ────────────────────────
    const { data: sub, error: subError } = await fetchSubscription(supabase, user.id);

    if (subError) {
      console.error(`[check-entitlement] DB error for ${user.id}:`, subError?.message);
      return jsonResponse({ allowed: false, reason: "Database error" });
    }
    // sub may be null for coins-only users who have no subscription row yet.
    // Treat as balance 0 / tier "none" — the capture function creates the row on purchase.

    const currentBalance: number = sub?.coins_balance ?? 0;

    // ── action: check_balance ─────────────────────────────────────────────────
    if (action === "check_balance") {
      return jsonResponse({
        allowed: true,
        balance: currentBalance,
        ...(sub?.tier   ? { tier:   sub.tier   } : {}),
        ...(sub?.status ? { status: sub.status } : {}),
        has_ever_purchased: sub?.has_ever_purchased === true,
      });
    }

    // ── action: deduct (TTS — client pre-calculates exact coin cost) ──────────
    if (action === "deduct") {
      const { feature, coins, prompt, model } = body;
      // Optional: caller can pass its own real computed cost (br-ai does,
      // via IMAGE_API_COST_USD) instead of relying on the MISC_USD guess
      // below, which only ever covered voice_clone/whisper_minute/
      // web_search — every brainrot_* feature fell through to `?? 0`,
      // which is why every BrainRot row in a ledger review showed a flat
      // $0.0000 api_cost_usd despite br-ai computing real costs internally
      // for pricing purposes; that number just never made it into the log.
      const callerCost = typeof body.api_cost_usd === "number" && body.api_cost_usd >= 0
        ? body.api_cost_usd
        : null;
      if (!feature || coins == null || coins < 0) {
        return jsonResponse({ allowed: false, reason: "Missing feature or coins" }, 400);
      }

      if (currentBalance < coins) {
        return jsonResponse({
          allowed: false,
          reason:  "Insufficient coins",
          balance: currentBalance,
          cost:    coins,
        });
      }

      const newBalance = currentBalance - coins;
      await supabase
        .from("subscriptions")
        .update({ coins_balance: newBalance, updated_at: new Date().toISOString() })
        .eq("user_id", user.id);

      // Log to both tables
      if (coins > 0) {
        await supabase.from("coin_transactions").insert({
          user_id:       user.id,
          amount:        coins,
          direction:     "debit",
          feature:       feature,
          description:   `Used ${feature}`,
          balance_after: newBalance,
        });
      }

      const apiCost     = callerCost ?? (MISC_USD[feature] ?? 0) * Math.max(1, coins / 10); // rough estimate when caller doesn't know its own cost
      const costPer100  = computeCostPer100Coins(apiCost, coins);
      const { data: logRow } = await supabase
        .from("ez_usage_log")
        .insert({
          user_id:            user.id,
          user_email:         userEmail,
          ip_address:         ipAddress,
          session_id:         body.session_id ?? null,
          feature,
          model:              model ?? null,
          prompt:             truncatePrompt(prompt),
          coins_charged:      coins,
          quantity:           1,
          running_balance:    newBalance,
          api_cost_usd:       apiCost.toFixed(6),
          cost_per_100_coins: costPer100,
          status:             "complete",
        })
        .select("id")
        .single();

      return jsonResponse({
        allowed: true,
        balance: newBalance,
        cost:    coins,
        ...(sub?.tier ? { tier: sub.tier } : {}),
        has_ever_purchased: sub?.has_ever_purchased === true,
        log_id:  logRow?.id ?? null,
      });
    }

    // ── action: check — flat-rate or token-based, deducts immediately ─────────
    if (action === "check") {
      const { feature, estimated_tokens, quantity, prompt, model } = body;
      if (!feature) {
        return jsonResponse({ allowed: false, reason: "No feature specified" }, 400);
      }

      const qty: number = (quantity != null && quantity > 1) ? Math.floor(quantity) : 1;

      let coinCost: number;
      let estimatedApiCost = 0;

      if (COIN_COSTS_PER_1K_TOKENS[feature] != null) {
        if (estimated_tokens == null || estimated_tokens <= 0) {
          return jsonResponse({
            allowed: false,
            reason:  "estimated_tokens required for this feature",
          }, 400);
        }
        coinCost = Math.ceil((estimated_tokens / 1000) * COIN_COSTS_PER_1K_TOKENS[feature]);
        // Use 70/30 split for initial estimate — corrected on refund_tokens callback
        const estInput  = Math.round(estimated_tokens * 0.70);
        const estOutput = Math.round(estimated_tokens * 0.30);
        estimatedApiCost = estimateChatApiCost(feature, model, estInput, estOutput);
      } else if (COIN_COSTS_FLAT[feature] != null) {
        coinCost = COIN_COSTS_FLAT[feature] * qty;
        // Use exact image pricing when model/quality/size are provided
        if (feature.startsWith("image") || feature.startsWith("dalle")) {
          const isEdit        = (body.is_edit === true);
          const promptTokens  = body.prompt_tokens ?? 80;
          estimatedApiCost    = estimateImageApiCost(
            feature, qty, model, body.img_quality, body.img_size,
            promptTokens, isEdit
          );
        } else if (feature.startsWith("sora")) {
          estimatedApiCost = estimateSoraApiCost(feature, model);
        } else {
          estimatedApiCost = (MISC_USD[feature] ?? 0) * qty;
        }
      } else {
        coinCost = 0;
      }

      if (currentBalance < coinCost) {
        return jsonResponse({
          allowed: false,
          reason:  "Insufficient coins",
          balance: currentBalance,
          cost:    coinCost,
        });
      }

      // FIXED (see this file's changelog): this used to deduct coinCost from
      // subscriptions.coins_balance, insert a real coin_transactions debit,
      // and — for every flat-rate feature (every image request, TTS, etc.)
      // — mark the ez_usage_log row "complete" immediately. All of that
      // directly contradicted this file's own header comment ("not the
      // source of truth for what gets charged") and meant every request
      // was charged twice: once here, unconditionally, on top of whatever
      // the actual edge function (ez-chat/ez-image/ez-elevenlabs) separately
      // and correctly deducts on its own. "check" now does exactly what its
      // name says: verifies affordability and returns, without touching
      // balance-related tables at all. The log row below is "pending"
      // regardless of feature type — there's no real completed transaction
      // to log yet at this point, for any feature.
      const { data: logRow } = await supabase
        .from("ez_usage_log")
        .insert({
          user_id:             user.id,
          user_email:          userEmail,
          ip_address:          ipAddress,
          session_id:          body.session_id ?? null,
          feature,
          model:               model   ?? null,
          prompt:              truncatePrompt(prompt),
          coins_charged:       coinCost, // estimate only — not a real charge, see above
          quantity:            qty,
          images_requested:    COIN_COSTS_FLAT[feature] && feature.startsWith("image") ? qty : null,
          running_balance:     currentBalance, // unchanged — this action doesn't deduct
          api_cost_usd:        estimatedApiCost.toFixed(6),
          cost_per_100_coins:  computeCostPer100Coins(estimatedApiCost, coinCost),
          status:              "pending",
        })
        .select("id")
        .single();

      console.log(`[check-entitlement] ${feature} qty=${qty} estimate=${coinCost}coins api_cost=$${estimatedApiCost.toFixed(4)} log_id=${logRow?.id} (read-only check, no balance change)`);

      return jsonResponse({
        allowed:  true,
        balance:  currentBalance,
        cost:     coinCost,
        ...(sub?.tier ? { tier: sub.tier } : {}),
        has_ever_purchased: sub?.has_ever_purchased === true,
        log_id:   logRow?.id ?? null,  // returned so client can update with actual data later
      });
    }

    return jsonResponse({ allowed: false, reason: "Unknown action" }, 400);

  } catch (e) {
    console.error("[check-entitlement] Uncaught error:", e);
    return jsonResponse({ allowed: false, reason: "Server error" }, 500);
  }
});
